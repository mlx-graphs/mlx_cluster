rejection_sampling ¶

rejection_sampling(
    rowptr: mlx.core.array,
    col: mlx.core.array,
    start: mlx.core.array,
    walk_length: int,
    p: float,
    q: float,
    *,
    key: object | None = None,
    max_attempts: int = 1000,
    stream: object | None = None,

node2vec biased random walks (rejection sampling).

Having moved t -> v, the next node x is a neighbor of v chosen with weight 1/p if x == t, 1 if x is a neighbor of t, and 1/q otherwise. A node without neighbors keeps the walk in place (edge id -1).

Random numbers are generated inside the kernel from a counter-based generator seeded by key, so the CPU and GPU produce identical walks for the same key.

def hetero_neighbor_sample(
    colptr_dict: Mapping[EdgeType, mx.array],
    row_dict: Mapping[EdgeType, mx.array],
    seed_dict: Mapping[NodeType, mx.array],
    num_neighbors: NumNeighbors,
    num_nodes_dict: Optional[Mapping[NodeType, int]] = None,
    replace: bool = False,
    directed: bool = True,
    seed: Optional[int] = None,
) -> HeteroSamplerOutput:
    """Multi-hop neighbor sampling on a heterogeneous graph (GraphSAGE style).

    For every edge type ``(src, rel, dst)`` and every hop, up to ``k``
    incoming ``src`` neighbors are sampled for each ``dst`` node discovered
    at the previous hop (the seeds at hop 0).

    Args:
        colptr_dict: CSC column pointers per edge type. For ``(src, rel,
            dst)`` this has ``num_dst_nodes + 1`` int64 entries.
        row_dict: CSC row indices (source node of each edge, sorted by
            destination) per edge type, int64.
        seed_dict: Seed node ids per node type. Must not contain duplicates.
        num_neighbors: Fan-out per hop, e.g. ``[10, 5]``. Either a single
            list used for every edge type, or a dict mapping each edge type
            to its own list. ``-1`` takes all neighbors; ``0`` samples none.
            Shorter lists are padded with ``0``.
        num_nodes_dict: Optional number of nodes per node type (e.g.
            ``HeteroGraphData.num_nodes``). Lets node types that only appear
            as a source use the faster dense id mapper.
        replace: Sample with replacement.
        directed: Only ``True`` is supported for heterogeneous graphs.
        seed: Seed for the sampler's RNG. If ``None``, one is drawn from
            MLX's global RNG, so ``mx.random.seed(...)`` makes sampling
            reproducible.

    Returns:
        HeteroSamplerOutput

    Example:

    .. code-block:: python

        out = hetero_neighbor_sample(
            colptr_dict={("author", "writes", "paper"): colptr},
            row_dict={("author", "writes", "paper"): row},
            seed_dict={"paper": mx.array([0, 1], dtype=mx.int64)},
            num_neighbors=[10, 5],
        )
        out.node_id["author"]                         # sampled author ids
        et = ("author", "writes", "paper")
        edge_index = mx.stack([out.row[et], out.col[et]])
    """
    edge_types = list(colptr_dict.keys())
    if not edge_types:
        raise ValueError("colptr_dict must contain at least one edge type")
    if set(row_dict.keys()) != set(edge_types):
        raise ValueError("colptr_dict and row_dict must have the same edge types")
    if not seed_dict:
        raise ValueError("seed_dict must contain at least one node type")

    rel_types = [edge_type_to_str(et) for et in edge_types]

    # Node types: everything mentioned by edge types, seeds or num_nodes_dict,
    # in first-seen order (deterministic output ordering).
    node_types: List[str] = []
    for et in edge_types:
        for nt in (et[0], et[2]):
            if nt not in node_types:
                node_types.append(nt)
    for nt in list(seed_dict.keys()) + list((num_nodes_dict or {}).keys()):
        if nt not in node_types:
            node_types.append(nt)

    fanouts = _resolve_num_neighbors(num_neighbors, edge_types)

    seeds = {}
    for nt, s in seed_dict.items():
        s = s if isinstance(s, mx.array) else mx.array(s)
        seeds[nt] = s.astype(mx.int64).reshape(-1)

    if seed is None:
        seed = int(mx.random.randint(0, 2**31 - 1, shape=()).item())

    node_id, row, col, edge_id, n_nodes, n_edges = _hetero_neighbor_sample(
        node_types,
        edge_types,
        {rt: colptr_dict[et] for rt, et in zip(rel_types, edge_types)},
        {rt: row_dict[et] for rt, et in zip(rel_types, edge_types)},
        seeds,
        {rt: fanouts[et] for rt, et in zip(rel_types, edge_types)},
        {k: int(v) for k, v in (num_nodes_dict or {}).items()},
        bool(replace),
        bool(directed),
        int(seed),
    )

    to_et = dict(zip(rel_types, edge_types))
    return HeteroSamplerOutput(
        node_id={nt: node_id[nt] for nt in node_types},
        row={to_et[rt]: row[rt] for rt in rel_types},
        col={to_et[rt]: col[rt] for rt in rel_types},
        edge_id={to_et[rt]: edge_id[rt] for rt in rel_types},
        num_sampled_nodes={nt: list(n_nodes[nt]) for nt in node_types},
        num_sampled_edges={to_et[rt]: list(n_edges[rt]) for rt in rel_types},
    )

def build_metapath_csr(
    edge_index_dict: Mapping[EdgeType, mx.array],
    metapath: Sequence[EdgeType],
    num_nodes_dict: Optional[Mapping[NodeType, int]] = None,
) -> MetaPathCSR:
    """Builds the CSR adjacency used by :func:`metapath_random_walk`.

    This sorts every relation once, so compute it once per graph and reuse it.

    Args:
        edge_index_dict: ``(src, rel, dst) -> [2, num_edges]`` edge indices,
            with node ids local to each node type (as in ``HeteroGraphData``).
        metapath: List of ``(src, rel, dst)`` edge types. Each edge type must
            end at the node type the next one starts from.
        num_nodes_dict: Optional number of nodes per node type. Missing types
            are inferred from the largest node id in ``edge_index_dict``.

    Returns:
        MetaPathCSR
    """
    metapath = _validate_metapath(metapath)
    missing = [et for et in metapath if et not in edge_index_dict]
    if missing:
        raise ValueError(f"Edge types {missing} are not in edge_index_dict")

    num_nodes = _infer_num_nodes(edge_index_dict, num_nodes_dict)

    unique_rels: List[EdgeType] = []
    for et in metapath:
        if et not in unique_rels:
            unique_rels.append(et)

    rowptrs, cols, rp_off, col_off = [], [], [], []
    rp_cursor, col_cursor = 0, 0
    for et in unique_rels:
        src_type = et[0]
        n_src = num_nodes.get(src_type, 0)
        ei = np.asarray(edge_index_dict[et]).astype(np.int64)
        src, dst = ei[0], ei[1]
        order = np.argsort(src, kind="stable")
        counts = np.bincount(src, minlength=n_src)
        rowptr = np.concatenate([[0], np.cumsum(counts)]).astype(np.int64)
        rowptrs.append(rowptr)
        cols.append(dst[order])
        rp_off.append(rp_cursor)
        col_off.append(col_cursor)
        rp_cursor += rowptr.shape[0]
        col_cursor += dst.shape[0]

    return MetaPathCSR(
        metapath=metapath,
        rowptr=mx.array(np.concatenate(rowptrs)),
        col=mx.array(np.concatenate(cols)),
        rowptr_offset=mx.array(np.array(rp_off, dtype=np.int64)),
        col_offset=mx.array(np.array(col_off, dtype=np.int64)),
        rel_ids=[unique_rels.index(et) for et in metapath],
        num_nodes_dict=num_nodes,
    )

def metapath_random_walk(
    csr: MetaPathCSR,
    start: mx.array,
    walk_length: int,
    rand: Optional[mx.array] = None,
    stream: Optional[mx.Stream] = None,
) -> Tuple[mx.array, mx.array]:
    """Meta-path guided uniform random walks.

    Step ``l`` of every walk follows edge type ``metapath[l % len(metapath)]``,
    choosing uniformly among the current node's neighbors of that type. If a
    node has no such neighbor, the walk stops and the remaining entries are
    ``-1``.

    Args:
        csr: Output of :func:`build_metapath_csr`.
        start: Start node ids, local to ``metapath[0][0]``'s node type.
        walk_length: Number of steps per walk. If it is longer than the
            meta-path, the meta-path must be cyclic (end where it starts).
        rand: Optional uniform ``[0, 1)`` values of shape
            ``[len(start), walk_length]``. Drawn from MLX's RNG if omitted,
            so ``mx.random.seed`` makes walks reproducible.
        stream: Stream or device to run on (``mx.cpu`` or ``mx.gpu``).

    Returns:
        tuple ``(nodes, edges)``: ``nodes`` is ``[len(start), walk_length + 1]``
        local node ids (the type of column ``j`` is
        ``metapath_node_types(metapath, walk_length)[j]``); ``edges`` is
        ``[len(start), walk_length]`` edge offsets within each relation's CSR.

    Example:

    .. code-block:: python

        metapath = [
            ("author", "writes", "paper"),
            ("paper", "published_in", "venue"),
            ("venue", "publishes", "paper"),
            ("paper", "written_by", "author"),
        ]
        csr = build_metapath_csr(edge_index_dict, metapath)
        nodes, _ = metapath_random_walk(csr, mx.arange(10), walk_length=8)
    """
    metapath = csr.metapath
    m = len(metapath)
    if walk_length < 1:
        raise ValueError("walk_length must be >= 1")
    if walk_length > m and metapath[-1][2] != metapath[0][0]:
        raise ValueError(
            f"walk_length ({walk_length}) is longer than the metapath ({m}), "
            "so the metapath must be cyclic: it ends at "
            f"'{metapath[-1][2]}' but starts at '{metapath[0][0]}'"
        )

    start = mx.array(start) if not isinstance(start, mx.array) else start
    start = start.astype(mx.int64).reshape(-1)
    if rand is None:
        rand = mx.random.uniform(shape=(start.shape[0], walk_length))
    rand = rand.astype(mx.float32)

    step_rel = mx.array(
        np.array([csr.rel_ids[l % m] for l in range(walk_length)], dtype=np.int64)
    )
    return _metapath_random_walk(
        csr.rowptr,
        csr.col,
        csr.rowptr_offset,
        csr.col_offset,
        step_rel,
        start,
        rand,
        walk_length,
        stream=stream,
    )

def metapath_node_types(metapath: Sequence[EdgeType], walk_length: int) -> List[str]:
    """Node type of each of the ``walk_length + 1`` positions of a walk."""
    m = len(metapath)
    return [metapath[0][0]] + [metapath[l % m][2] for l in range(walk_length)]

def coo_to_csr(
    row: mx.array,
    col: mx.array,
    num_rows: int,
    values: Optional[mx.array] = None,
    long_row_threshold: Optional[int] = 256,
    chunk_size: int = 2048,
    transpose: bool = True,
    num_cols: Optional[int] = None,
) -> CSR:
    """Converts a COO edge list into CSR (sorted by ``row``).

    This needs a sort, so compute it once per graph and reuse it.

    Args:
        long_row_threshold: rows with more edges than this get the
            load-balanced long-row path in :func:`spmm_csr` (see
            :class:`LongRows`). ``None`` disables it and keeps the conversion
            fully lazy (no host read-back), e.g. inside ``mx.compile``.
        chunk_size: maximum number of edges per long-row chunk.
        transpose: also build the transposed CSR (one more sort, ~3 extra
            index arrays) so the backward pass of :func:`spmm_csr` uses the
            CSR kernel: deterministic and atomic-free. Set ``False`` to save
            the memory when you only need the forward pass.
        num_cols: number of columns (source nodes), i.e. ``x.shape[0]`` in
            :func:`spmm_csr`. Defaults to ``num_rows`` (square adjacency).
    """
    row = row.astype(mx.int64)
    perm = mx.argsort(row)
    # int32 counts: MLX's GPU scatter does not support int64 inputs/updates.
    counts = mx.zeros((num_rows,), dtype=mx.int32).at[row].add(1)
    rowptr = mx.concatenate(
        [mx.zeros((1,), dtype=mx.int64), mx.cumsum(counts.astype(mx.int64))]
    )
    long_rows = (
        None
        if long_row_threshold is None
        else build_long_rows(rowptr, long_row_threshold, chunk_size)
    )
    col_csr = col.astype(mx.int64)[perm]

    csr_t = None
    if transpose:
        n_cols = num_rows if num_cols is None else num_cols
        perm_t = mx.argsort(col_csr)  # CSR edge index of each transposed edge
        counts_t = mx.zeros((n_cols,), dtype=mx.int32).at[col_csr].add(1)
        rowptr_t = mx.concatenate(
            [mx.zeros((1,), dtype=mx.int64), mx.cumsum(counts_t.astype(mx.int64))]
        )
        csr_t = CSRTranspose(
            rowptr=rowptr_t,
            col=row[perm][perm_t],
            perm=perm_t.astype(mx.int64),
            long_rows=(
                None
                if long_row_threshold is None
                else build_long_rows(rowptr_t, long_row_threshold, chunk_size)
            ),
            edge=perm[perm_t],
        )

    return CSR(
        rowptr=rowptr,
        col=col_csr,
        values=None if values is None else values[perm],
        perm=perm,
        long_rows=long_rows,
        transpose=csr_t,
    )

def build_long_rows(
    rowptr: mx.array, threshold: int = 256, chunk_size: int = 2048
) -> Optional[LongRows]:
    """Finds rows with more than ``threshold`` edges and splits them into
    chunks of at most ``chunk_size`` edges. Returns ``None`` if there are none.

    Runs once per graph (it reads ``rowptr`` back to the host).
    """
    if threshold < 0 or chunk_size < 1:
        raise ValueError("threshold must be >= 0 and chunk_size >= 1")
    rp = np.array(rowptr, dtype=np.int64)
    deg = np.diff(rp)
    rows = np.nonzero(deg > threshold)[0]
    if rows.size == 0:
        return None
    n_chunks = (deg[rows] + chunk_size - 1) // chunk_size
    chunk_ptr = np.concatenate([[0], np.cumsum(n_chunks)]).astype(np.int64)
    owner = np.repeat(rows, n_chunks)  # long row of each chunk
    local = np.arange(chunk_ptr[-1]) - np.repeat(chunk_ptr[:-1], n_chunks)
    start = rp[owner] + local * chunk_size
    end = np.minimum(start + chunk_size, rp[owner + 1])
    return LongRows(
        rows=mx.array(rows.astype(np.int64)),
        chunk_ptr=mx.array(chunk_ptr),
        chunk_start=mx.array(start.astype(np.int64)),
        chunk_end=mx.array(end.astype(np.int64)),
        threshold=int(threshold),
        owner=mx.array(np.repeat(np.arange(rows.size, dtype=np.int64), n_chunks)),
    )

LongRows ¶

Bases: NamedTuple

Load-balancing metadata for CSR rows with many edges.

Rows with more than threshold edges are not handled by the regular one-thread-per-(row, feature) CSR kernel, where a single thread would loop over every edge of a hub node. Instead each such row is cut into chunks of at most chunk_size consecutive edges, each chunk is summed by its own GPU threadgroup, and the chunk sums are added in a fixed order (results stay deterministic). Built by coo_to_csr().
